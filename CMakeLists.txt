cmake_minimum_required(VERSION 3.20)
project(doubao_ime_linux VERSION 0.1.0 LANGUAGES CXX)
find_package(Fcitx5Core REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(JSONC REQUIRED IMPORTED_TARGET json-c)
add_library(doubaoime MODULE fcitx5/keyboard.cpp)
target_link_libraries(doubaoime PRIVATE Fcitx5::Core PkgConfig::JSONC)
option(DOUBAO_BUILD_VOICE "Build optional Doubao dictation endpoint" ON)
if(DOUBAO_BUILD_VOICE)
  find_package(Fcitx5Module REQUIRED COMPONENTS DBus)
  add_library(doubaovoice MODULE fcitx5/voice.cpp)
  target_link_libraries(doubaovoice PRIVATE Fcitx5::Core Fcitx5::Module::DBus)
endif()
foreach(target doubaoime doubaovoice)
  if(TARGET ${target})
    target_compile_features(${target} PRIVATE cxx_std_17)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    set_target_properties(${target} PROPERTIES PREFIX "")
  endif()
endforeach()
